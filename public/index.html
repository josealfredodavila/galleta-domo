// ================================================================ // WALLET - CON METAMASK (CON DESCARGA) // ================================================================ let web3 = null; let userAccount = null; async function conectarWallet() { const connectBtn = document.getElementById('connectWallet'); // ===== VERIFICAR SI METAMASK ESTÁ INSTALADO ===== if (!window.ethereum) { connectBtn.disabled = true; connectBtn.textContent = '⬇️ Descargar MetaMask'; connectBtn.style.background = 'linear-gradient(135deg, #ff6b35, #f7931a)'; connectBtn.style.color = 'white'; connectBtn.style.border = 'none'; connectBtn.style.boxShadow = '0 4px 20px rgba(255, 107, 53, 0.3)'; const descargar = confirm( '⚠️ MetaMask no está instalado.\n\n' + 'Sariel\'s funciona con MetaMask para conectar tu wallet.\n\n' + '¿Quieres ir a la página oficial de MetaMask para descargarlo?' ); if (descargar) { window.open('https://metamask.io/download/', '_blank'); } else { showToast('⚠️ Instala MetaMask para continuar', 'warning'); connectBtn.disabled = false; connectBtn.textContent = '◈ Conectar Wallet'; connectBtn.style.background = ''; connectBtn.style.color = ''; connectBtn.style.boxShadow = ''; } return; } // ===== METAMASK INSTALADO ===== try { // Verificar si ya hay cuentas conectadas const accounts = await window.ethereum.request({ method: 'eth_accounts' }); if (accounts && accounts.length > 0) { userAccount = accounts[0]; actualizarUIWallet(userAccount); showToast('✅ Wallet ya conectada'); return; } } catch (error) { console.warn('Error verificando cuentas:', error); } // Solicitar conexión try { connectBtn.disabled = true; connectBtn.textContent = '⏳ Conectando...'; const newAccounts = await window.ethereum.request({ method: 'eth_requestAccounts' }); if (newAccounts && newAccounts.length > 0) { userAccount = newAccounts[0]; actualizarUIWallet(userAccount); showToast('✅ Wallet conectada'); } } catch (error) { console.error('Error conectando wallet:', error); if (error.code === 4001) { showToast('❌ Conexión rechazada por el usuario', 'error'); } else { showToast('❌ Error al conectar wallet', 'error'); } } finally { connectBtn.disabled = false; connectBtn.textContent = '◈ Conectar Wallet'; connectBtn.style.background = ''; connectBtn.style.color = ''; connectBtn.style.boxShadow = ''; } } function actualizarUIWallet(wallet) { document.getElementById('connectWallet').style.display = 'none'; document.getElementById('walletStatusBar').style.display = 'flex'; document.getElementById('walletAddressDisplay').textContent = wallet.slice(0, 6) + '...' + wallet.slice(-4); document.getElementById('walletStatusText').textContent = 'Conectado'; document.getElementById('buyDomo').disabled = false; document.getElementById('buyDomoCrypto').disabled = false; localStorage.setItem('sariels_wallet', wallet); } function desconectarWallet() { userAccount = null; localStorage.removeItem('sariels_wallet'); document.getElementById('connectWallet').style.display = 'inline-flex'; document.getElementById('connectWallet').disabled = false; document.getElementById('connectWallet').textContent = '◈ Conectar Wallet'; document.getElementById('connectWallet').style.background = ''; document.getElementById('connectWallet').style.color = ''; document.getElementById('connectWallet').style.boxShadow = ''; document.getElementById('walletStatusBar').style.display = 'none'; document.getElementById('buyDomo').disabled = true; document.getElementById('buyDomoCrypto').disabled = true; showToast('◈ Wallet desconectada'); }

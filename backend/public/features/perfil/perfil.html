/** * routes/mensajes.js * API de Mensajería para Sariel's * * Endpoints: * GET /api/mensajes/conversaciones - Lista de conversaciones del usuario * GET /api/mensajes/mensajes/:contactoId - Mensajes con un contacto * POST /api/mensajes/mensajes - Enviar nuevo mensaje * PATCH /api/mensajes/mensajes/leer - Marcar mensajes como leídos * DELETE /api/mensajes/mensajes/:mensajeId - Eliminar mensaje * PUT /api/mensajes/mensajes/:mensajeId - Editar mensaje * DELETE /api/mensajes/conversaciones/:contactoId - Eliminar conversación * POST /api/mensajes/reportar/:mensajeId - Reportar mensaje * POST /api/mensajes/bloquear/:usuarioId - Bloquear usuario */ const express = require('express'); const router = express.Router(); const { createClient } = require('@supabase/supabase-js'); // --- Configuración desde variables de entorno --- const supabaseUrl = process.env.SUPABASE_URL; const supabaseAnonKey = process.env.SUPABASE_ANON_KEY; const supabaseServiceKey = process.env.SUPABASE_SERVICE_ROLE_KEY; if (!supabaseUrl || !supabaseAnonKey) { console.error('❌ Error: Variables de entorno SUPABASE_URL y SUPABASE_ANON_KEY son requeridas'); process.exit(1); } // Cliente público para validar tokens const supabasePublic = createClient(supabaseUrl, supabaseAnonKey); // Cliente con Service Role para operaciones de base de datos (se usa en endpoints que requieren permisos elevados) const supabaseAdmin = supabaseServiceKey ? createClient(supabaseUrl, supabaseServiceKey, { auth: { autoRefreshToken: false, persistSession: false } }) : supabasePublic; // --- Middleware: Verificar autenticación --- async function verificarAuth(req, res, next) { try { const authHeader = req.headers.authorization; if (!authHeader || !authHeader.startsWith('Bearer ')) { return res.status(401).json({ error: 'Token de autenticación requerido' }); } const token = authHeader.split(' ')[1]; if (!token) { return res.status(401).json({ error: 'Token inválido' }); } // Verificar token con Supabase const { data: { user }, error } = await supabasePublic.auth.getUser(token); if (error || !user) { console.warn('⚠️ Auth error:', error?.message); return res.status(401).json({ error: 'No autorizado', details: error?.message }); } // Adjuntar usuario a la request req.user = user; req.userId = user.id; // También guardamos el token para futuras llamadas seguras req.accessToken = token; next(); } catch (error) { console.error('❌ Error en verificarAuth:', error); return res.status(500).json({ error: 'Error interno de autenticación' }); } } // --- Helper: Crear cliente de Supabase con token del usuario --- function getSupabaseClient(accessToken) { if (!accessToken) return supabaseAdmin; return createClient(supabaseUrl, supabaseAnonKey, { auth: { autoRefreshToken: false, persistSession: false }, global: { headers: { Authorization: `Bearer ${accessToken}` } } }); } // --- RUTAS --- /** * GET /api/mensajes/conversaciones * Obtiene todas las conversaciones del usuario autenticado */ router.get('/conversaciones', verificarAuth, async (req, res) => { try { const userId = req.userId; const supabase = getSupabaseClient(req.accessToken); // Obtener conversaciones: mensajes agrupados por contacto // 1. Obtener todos los mensajes del usuario (enviados y recibidos) const { data: mensajes, error } = await supabase .from('mensajes_chat') .select(` id, remitente_id, destinatario_id, contenido, imagen_url, tipo, mime_type, leido, created_at, editado, eliminado, eliminado_at, remitente:remitente_id(id, nombre, handle, avatar_url), destinatario:destinatario_id(id, nombre, handle, avatar_url) `) .or(`remitente_id.eq.${userId},destinatario_id.eq.${userId}`) .order('created_at', { ascending: false }); if (error) { console.error('❌ Error en conversaciones:', error); return res.status(500).json({ error: 'Error al obtener conversaciones', details: error.message }); } if (!mensajes || mensajes.length === 0) { return res.json({ data: [] }); } // Agrupar por contacto (la otra persona) const conversacionesMap = new Map(); for (const msg of mensajes) { const contactoId = msg.remitente_id === userId ? msg.destinatario_id : msg.remitente_id; const contacto = msg.remitente_id === userId ? msg.destinatario : msg.remitente; if (!conversacionesMap.has(contactoId)) { conversacionesMap.set(contactoId, { contacto_id: contactoId, contacto: contacto || { id: contactoId, nombre: 'Usuario', handle: null }, ultimo_mensaje: msg.contenido || '[Archivo]', ultimo_mensaje_fecha: msg.created_at, mensajes_no_leidos: 0, mensajes: [] }); } const conv = conversacionesMap.get(contactoId); conv.mensajes.push(msg); // Contar no leídos (mensajes que no son del usuario y no están leídos) if (msg.remitente_id !== userId && !msg.leido) { conv.mensajes_no_leidos += 1; } } // Convertir a array y ordenar por último mensaje const conversaciones = Array.from(conversacionesMap.values()) .map(conv => ({ ...conv, // Tomar el mensaje más reciente como último ultimo_mensaje: conv.mensajes[0]?.contenido || '[Archivo]', ultimo_mensaje_fecha: conv.mensajes[0]?.created_at, mensajes_no_leidos: conv.mensajes_no_leidos })) .sort((a, b) => new Date(b.ultimo_mensaje_fecha) - new Date(a.ultimo_mensaje_fecha)) .map(({ mensajes, ...rest }) => rest); // Remover array de mensajes res.json({ data: conversaciones }); } catch (error) { console.error('❌ Error en GET /conversaciones:', error); res.status(500).json({ error: 'Error interno del servidor' }); } }); /** * GET /api/mensajes/mensajes/:contactoId * Obtiene los mensajes con un contacto específico */ router.get('/mensajes/:contactoId', verificarAuth, async (req, res) => { try { const userId = req.userId; const { contactoId } = req.params; const { search } = req.query; if (!contactoId) { return res.status(400).json({ error: 'ID de contacto requerido' }); } const supabase = getSupabaseClient(req.accessToken); let query = supabase .from('mensajes_chat') .select(` id, remitente_id, destinatario_id, contenido, imagen_url, tipo, mime_type, leido, created_at, editado, eliminado, eliminado_at `) .or(`and(remitente_id.eq.${userId},destinatario_id.eq.${contactoId}),and(remitente_id.eq.${contactoId},destinatario_id.eq.${userId})`) .order('created_at', { ascending: true }); // Búsqueda por contenido si se proporciona if (search && search.trim().length > 0) { query = query.ilike('contenido', `%${search.trim()}%`); } const { data, error } = await query; if (error) { console.error('❌ Error en mensajes:', error); return res.status(500).json({ error: 'Error al obtener mensajes', details: error.message }); } res.json({ data: data || [] }); } catch (error) { console.error('❌ Error en GET /mensajes/:contactoId:', error); res.status(500).json({ error: 'Error interno del servidor' }); } }); /** * POST /api/mensajes/mensajes * Envía un nuevo mensaje */ router.post('/mensajes', verificarAuth, async (req, res) => { try { const userId = req.userId; const { destinatario_id, contenido, imagen_url, tipo, mime_type } = req.body; if (!destinatario_id) { return res.status(400).json({ error: 'Destinatario requerido' }); } if (!contenido && !imagen_url) { return res.status(400).json({ error: 'Se requiere contenido o archivo' }); } // Validar que el destinatario existe const supabase = getSupabaseClient(req.accessToken); const { data: usuario, error: userError } = await supabase .from('usuarios') .select('id') .eq('id', destinatario_id) .single(); if (userError || !usuario) { return res.status(404).json({ error: 'Destinatario no encontrado' }); } // Verificar que el usuario no esté bloqueado const { data: bloqueado, error: blockError } = await supabase .from('bloqueos') .select('id') .or(`and(bloqueador_id.eq.${userId},bloqueado_id.eq.${destinatario_id}),and(bloqueador_id.eq.${destinatario_id},bloqueado_id.eq.${userId})`) .maybeSingle(); if (bloqueado) { return res.status(403).json({ error: 'No puedes enviar mensajes a este usuario' }); } // Crear mensaje const nuevoMensaje = { remitente_id: userId, destinatario_id, contenido: contenido || '', imagen_url: imagen_url || null, tipo: tipo || 'texto', mime_type: mime_type || null, leido: false, editado: false, eliminado: false, created_at: new Date().toISOString() }; const { data, error } = await supabase .from('mensajes_chat') .insert(nuevoMensaje) .select() .single(); if (error) { console.error('❌ Error al enviar mensaje:', error); return res.status(500).json({ error: 'Error al enviar mensaje', details: error.message }); } res.status(201).json({ data }); } catch (error) { console.error('❌ Error en POST /mensajes:', error); res.status(500).json({ error: 'Error interno del servidor' }); } }); /** * PATCH /api/mensajes/mensajes/leer * Marca todos los mensajes de una conversación como leídos */ router.patch('/mensajes/leer', verificarAuth, async (req, res) => { try { const userId = req.userId; const { contacto_id } = req.body; if (!contacto_id) { return res.status(400).json({ error: 'ID de contacto requerido' }); } const supabase = getSupabaseClient(req.accessToken); // Actualizar mensajes no leídos del contacto que han sido enviados al usuario const { data, error } = await supabase .from('mensajes_chat') .update({ leido: true }) .eq('remitente_id', contacto_id) .eq('destinatario_id', userId) .eq('leido', false) .select(); if (error) { console.error('❌ Error al marcar leídos:', error); return res.status(500).json({ error: 'Error al marcar mensajes como leídos', details: error.message }); } res.json({ data: { actualizados: data?.length || 0 } }); } catch (error) { console.error('❌ Error en PATCH /mensajes/leer:', error); res.status(500).json({ error: 'Error interno del servidor' }); } }); /** * DELETE /api/mensajes/mensajes/:mensajeId * Elimina un mensaje (soft delete) */ router.delete('/mensajes/:mensajeId', verificarAuth, async (req, res) => { try { const userId = req.userId; const { mensajeId } = req.params; if (!mensajeId) { return res.status(400).json({ error: 'ID de mensaje requerido' }); } const supabase = getSupabaseClient(req.accessToken); // Verificar que el mensaje existe y pertenece al usuario const { data: mensaje, error: findError } = await supabase .from('mensajes_chat') .select('id, remitente_id') .eq('id', mensajeId) .single(); if (findError || !mensaje) { return res.status(404).json({ error: 'Mensaje no encontrado' }); } if (mensaje.remitente_id !== userId) { return res.status(403).json({ error: 'No puedes eliminar este mensaje' }); } // Soft delete const { data, error } = await supabase .from('mensajes_chat') .update({ eliminado: true, eliminado_at: new Date().toISOString(), contenido: null, imagen_url: null }) .eq('id', mensajeId) .select() .single(); if (error) { console.error('❌ Error al eliminar mensaje:', error); return res.status(500).json({ error: 'Error al eliminar mensaje', details: error.message }); } res.json({ data }); } catch (error) { console.error('❌ Error en DELETE /mensajes/:mensajeId:', error); res.status(500).json({ error: 'Error interno del servidor' }); } }); /** * PUT /api/mensajes/mensajes/:mensajeId * Edita un mensaje existente */ router.put('/mensajes/:mensajeId', verificarAuth, async (req, res) => { try { const userId = req.userId; const { mensajeId } = req.params; const { contenido } = req.body; if (!mensajeId) { return res.status(400).json({ error: 'ID de mensaje requerido' }); } if (!contenido || contenido.trim().length === 0) { return res.status(400).json({ error: 'El contenido no puede estar vacío' }); } const supabase = getSupabaseClient(req.accessToken); // Verificar que el mensaje existe y pertenece al usuario const { data: mensaje, error: findError } = await supabase .from('mensajes_chat') .select('id, remitente_id, eliminado') .eq('id', mensajeId) .single(); if (findError || !mensaje) { return res.status(404).json({ error: 'Mensaje no encontrado' }); } if (mensaje.remitente_id !== userId) { return res.status(403).json({ error: 'No puedes editar este mensaje' }); } if (mensaje.eliminado) { return res.status(400).json({ error: 'No puedes editar un mensaje eliminado' }); } // Actualizar mensaje const { data, error } = await supabase .from('mensajes_chat') .update({ contenido: contenido.trim(), editado: true, editado_at: new Date().toISOString() }) .eq('id', mensajeId) .select() .single(); if (error) { console.error('❌ Error al editar mensaje:', error); return res.status(500).json({ error: 'Error al editar mensaje', details: error.message }); } res.json({ data }); } catch (error) { console.error('❌ Error en PUT /mensajes/:mensajeId:', error); res.status(500).json({ error: 'Error interno del servidor' }); } }); /** * DELETE /api/mensajes/conversaciones/:contactoId * Elimina toda la conversación con un contacto */ router.delete('/conversaciones/:contactoId', verificarAuth, async (req, res) => { try { const userId = req.userId; const { contactoId } = req.params; if (!contactoId) { return res.status(400).json({ error: 'ID de contacto requerido' }); } const supabase = getSupabaseClient(req.accessToken); // Soft delete de todos los mensajes de la conversación const { data, error } = await supabase .from('mensajes_chat') .update({ eliminado: true, eliminado_at: new Date().toISOString(), contenido: null, imagen_url: null }) .or(`and(remitente_id.eq.${userId},destinatario_id.eq.${contactoId}),and(remitente_id.eq.${contactoId},destinatario_id.eq.${userId})`) .select(); if (error) { console.error('❌ Error al eliminar conversación:', error); return res.status(500).json({ error: 'Error al eliminar conversación', details: error.message }); } res.json({ data: { eliminados: data?.length || 0 } }); } catch (error) { console.error('❌ Error en DELETE /conversaciones/:contactoId:', error); res.status(500).json({ error: 'Error interno del servidor' }); } }); /** * POST /api/mensajes/reportar/:mensajeId * Reporta un mensaje */ router.post('/reportar/:mensajeId', verificarAuth, async (req, res) => { try { const userId = req.userId; const { mensajeId } = req.params; const { motivo } = req.body; if (!mensajeId) { return res.status(400).json({ error: 'ID de mensaje requerido' }); } if (!motivo || motivo.trim().length < 3) { return res.status(400).json({ error: 'Motivo de reporte requerido (mínimo 3 caracteres)' }); } const supabase = getSupabaseClient(req.accessToken); // Verificar que el mensaje existe const { data: mensaje, error: findError } = await supabase .from('mensajes_chat') .select('id, remitente_id, destinatario_id') .eq('id', mensajeId) .single(); if (findError || !mensaje) { return res.status(404).json({ error: 'Mensaje no encontrado' }); } // Verificar que el usuario es parte de la conversación if (mensaje.remitente_id !== userId && mensaje.destinatario_id !== userId) { return res.status(403).json({ error: 'No puedes reportar este mensaje' }); } // Crear reporte const reporte = { mensaje_id: mensajeId, reportado_por: userId, motivo: motivo.trim(), estado: 'pendiente', created_at: new Date().toISOString() }; const { data, error } = await supabase .from('reportes_mensajes') .insert(reporte) .select() .single(); if (error) { console.error('❌ Error al reportar mensaje:', error); return res.status(500).json({ error: 'Error al reportar mensaje', details: error.message }); } res.status(201).json({ data }); } catch (error) { console.error('❌ Error en POST /reportar/:mensajeId:', error); res.status(500).json({ error: 'Error interno del servidor' }); } }); /** * POST /api/mensajes/bloquear/:usuarioId * Bloquea a un usuario */ router.post('/bloquear/:usuarioId', verificarAuth, async (req, res) => { try { const userId = req.userId; const { usuarioId } = req.params; if (!usuarioId) { return res.status(400).json({ error: 'ID de usuario requerido' }); } if (usuarioId === userId) { return res.status(400).json({ error: 'No puedes bloquearte a ti mismo' }); } const supabase = getSupabaseClient(req.accessToken); // Verificar que el usuario existe const { data: usuario, error: userError } = await supabase .from('usuarios') .select('id') .eq('id', usuarioId) .single(); if (userError || !usuario) { return res.status(404).json({ error: 'Usuario no encontrado' }); } // Verificar si ya está bloqueado const { data: existente, error: existError } = await supabase .from('bloqueos') .select('id') .eq('bloqueador_id', userId) .eq('bloqueado_id', usuarioId) .maybeSingle(); if (existente) { return res.status(400).json({ error: 'Este usuario ya está bloqueado' }); } // Crear bloqueo const bloqueo = { bloqueador_id: userId, bloqueado_id: usuarioId, created_at: new Date().toISOString() }; const { data, error } = await supabase .from('bloqueos') .insert(bloqueo) .select() .single(); if (error) { console.error('❌ Error al bloquear usuario:', error); return res.status(500).json({ error: 'Error al bloquear usuario', details: error.message }); } // Eliminar mensajes de la conversación (soft delete) await supabase .from('mensajes_chat') .update({ eliminado: true, eliminado_at: new Date().toISOString(), contenido: null, imagen_url: null }) .or(`and(remitente_id.eq.${userId},destinatario_id.eq.${usuarioId}),and(remitente_id.eq.${usuarioId},destinatario_id.eq.${userId})`); res.status(201).json({ data }); } catch (error) { console.error('❌ Error en POST /bloquear/:usuarioId:', error); res.status(500).json({ error: 'Error interno del servidor' }); } }); module.exports = router;

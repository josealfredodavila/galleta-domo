/* ============================================ CHAT TEXTO BOT ============================================ */ async function preguntarAlBot(mensaje){ const s=await session(); if(!s){ throw new Error( 'No hay una sesión activa. Inicia sesión nuevamente.' ); } const controller= new AbortController(); const timeout= setTimeout( ()=>{ controller.abort(); }, 50000 ); try{ const r=await fetch( '/api/ai/chat', { method:'POST', headers:{ 'Content-Type': 'application/json', 'Accept': 'application/json', 'Authorization': 'Bearer '+s.access_token }, body:JSON.stringify({ message: mensaje.trim(), context: 'chat_sariels', history:[] }), signal: controller.signal } ); let data={}; try{ data= await r.json(); }catch(jsonError){ console.error( '❌ Marquinhos devolvió una respuesta que no es JSON:', jsonError ); throw new Error( 'El servidor de Marquinhos devolvió una respuesta inválida.' ); } if(!r.ok){ console.error( '❌ Error HTTP Marquinhos:', { status:r.status, data:data } ); throw new Error( data.error || `Error del servidor (${r.status})` ); } if(!data.success){ throw new Error( data.error || 'Marquinhos no pudo responder.' ); } const respuesta= typeof data.reply==='string' ? data.reply.trim() : ''; if(!respuesta){ throw new Error( 'Marquinhos respondió sin contenido.' ); } return respuesta; }catch(error){ if( error.name==='AbortError' ){ console.error( '⏱️ Timeout esperando a Marquinhos' ); throw new Error( 'Marquinhos está tardando demasiado en responder. Intenta nuevamente.' ); } console.error( '❌ Error comunicando con Marquinhos:', error ); throw error; }finally{ clearTimeout( timeout ); } }
